@startuml
!theme plain

title Diagrama Entidad-Relación - Plataforma de Reciclaje Temuco

entity "Usuario" as usuario {
  *id_usuario : int
  *nombre : string
  *correo : string
  *contraseña : string
  tipo_usuario : string  // común o emprendedor
  verificado : boolean
}

entity "PuntoVerde" as puntoVerde {
  *id_punto : int
  *nombre : string
  *ubicacion : string
  estado : string // activo, inactivo, en mal estado
  descripcion : string
}

entity "Comentario" as comentario {
  *id_comentario : int
  contenido : string
  fecha : date
}

entity "Notificación" as notificacion {
  *id_notificacion : int
  titulo : string
  contenido : string
  fecha : date
  tipo : string // evento, aviso municipal, charla, etc.
}

entity "GuíaAprendizaje" as guia {
  *id_guia : int
  titulo : string
  tipo_recurso : string // video, infografía
  enlace : string
}

entity "Calendario" as calendario {
  *id_evento : int
  titulo : string
  fecha : date
  descripcion : string
  tipo_evento : string // feria, voluntariado, charla, etc.
}

entity "Residuo" as residuo {
  *id_residuo : int
  nombre : string
  tipo : string // plástico, vidrio, papel, etc.
  descripcion : string
}

entity "Transacción" as transaccion {
  *id_transaccion : int
  tipo : string // compra o venta
  fecha : date
  cantidad : int
  precio : float
}

'-------------------------------
' Relaciones
'-------------------------------

usuario ||--o{ comentario : "realiza"
comentario }o--|| puntoVerde : "sobre"

usuario ||--o{ notificacion : "publica"
usuario ||--o{ guia : "crea"
usuario ||--o{ calendario : "organiza"

usuario ||--o{ transaccion : "participa en"
transaccion }o--|| residuo : "involucra"

puntoVerde ||--o{ residuo : "acepta"
usuario ||--o{ puntoVerde : "agrega/verifica"

puntoVerde ||--o{ calendario : "se asocia con evento"

@enduml
