'=========================================
' CLASES PRINCIPALES
'=========================================

class Usuario {
  - id_usuario : int
  - nombre : string
  - correo : string
  - contraseña : string
  - tipo_usuario : string // común o emprendedor
  - verificado : boolean
  --
  + registrarse()
  + iniciarSesion()
  + cerrarSesion()
  + editarPerfil()
  + recibirNotificacion()
}

class PuntoVerde {
  - id_punto : int
  - nombre : string
  - ubicacion : string
  - estado : string // activo, inactivo, en mal estado
  - descripcion : string
  --
  + mostrarDetalles()
  + actualizarEstado()
  + agregarComentario()
}

class Comentario {
  - id_comentario : int
  - contenido : string
  - fecha : date
  --
  + publicar()
  + editar()
  + eliminar()
}

class Notificacion {
  - id_notificacion : int
  - titulo : string
  - contenido : string
  - fecha : date
  - tipo : string // evento, aviso municipal, etc.
  --
  + publicar()
  + eliminar()
}

class GuiaAprendizaje {
  - id_guia : int
  - titulo : string
  - tipo_recurso : string // video, infografía
  - enlace : string
  --
  + mostrarGuia()
}

class Calendario {
  - id_evento : int
  - titulo : string
  - fecha : date
  - descripcion : string
  - tipo_evento : string // feria, voluntariado, charla, etc.
  --
  + agregarEvento()
  + eliminarEvento()
  + mostrarEventos()
}

class Residuo {
  - id_residuo : int
  - nombre : string
  - tipo : string
  - descripcion : string
  --
  + mostrarInformacion()
}

class Transaccion {
  - id_transaccion : int
  - tipo : string // compra o venta
  - fecha : date
  - cantidad : int
  - precio : float
  --
  + realizarTransaccion()
  + cancelarTransaccion()
}

'=========================================
' RELACIONES ENTRE CLASES
'=========================================

Usuario "1" -- "0..*" PuntoVerde : agrega/verifica >
Usuario "1" -- "0..*" Comentario : realiza >
PuntoVerde "1" -- "0..*" Comentario : recibe >
Usuario "1" -- "0..*" Notificacion : publica >
Usuario "1" -- "0..*" GuiaAprendizaje : crea >
Usuario "1" -- "0..*" Calendario : organiza >
Usuario "1" -- "0..*" Transaccion : participa >
Transaccion "*" -- "1" Residuo : involucra >
PuntoVerde "1" -- "0..*" Residuo : acepta >
PuntoVerde "1" -- "0..*" Calendario : se_asocia_con >

@enduml
